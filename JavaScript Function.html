function add(a, b) { return a + b; } function subtract(a, b) { return a - b; } function multiply(a, b) { return a * b; } function divide(a, b) { if (b === 0) { return "Cannot divide by zero"; } return a / b; } console.log("Addition:", add(10, 5)); console.log("Subtraction:", subtract(10, 5)); console.log("Multiplication:", multiply(10, 5)); console.log("Division:", divide(10, 5));
